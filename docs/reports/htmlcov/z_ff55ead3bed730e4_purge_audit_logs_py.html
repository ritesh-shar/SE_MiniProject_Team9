27 def handle(self, *args, **options):

31 before_str = options.get('before')

36 cutoff = timezone.make_aware(datetime.combine(parsed, datetime.min.time()))

42 else: 

43 cutoff = statutory_limit 

44 

45 # Delete only records strictly older than cutoff 

46 deleted_count, _ = AuditLog.objects.filter(timestamp__lt=cutoff).delete() 

47 self.stdout.write( 

48 self.style.SUCCESS(
